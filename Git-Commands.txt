GIT COMMANDS - BASIC TO ADVANCED | HINGLISH PRACTICAL REFERENCE

sudo dnf install git # RHEL/Fedora par Git install karta hai

git config --local --edit # Repository config editor mein kholta hai

git config --global core.autocrlf input # Commit par CRLF ko LF mein normalize karta hai

07. REMOTES, FETCH, PULL AND PUSH

git ls-remote --heads origin # Remote branches aur commit IDs list karta hai

git push -u origin main # Main push karta hai aur upstream set karta hai

08. MERGE AND CONFLICTS
git switch main # Target branch main select karta hai
git merge feature/nginx # Feature history current branch mein integrate karta hai
git merge --no-ff feature/nginx # Fast-forward possible hone par bhi merge commit banata hai
git merge --ff-only feature/nginx # Diverged history hone par merge reject karta hai
git merge --squash feature/nginx # Combined changes stage karta hai, merge commit nahi banata
git merge --no-commit --no-ff feature/nginx # Merge commit se pehle review ka mauka deta hai
git diff --name-only --diff-filter=U # Unresolved conflict files list karta hai
git ls-files -u # Conflicted index entries aur stages dikhata hai
git checkout --ours file.txt # Merge conflict mein current side ka version working tree mein rakhta hai
git checkout --theirs file.txt # Merge conflict mein incoming side ka version rakhta hai
git add file.txt # Manually resolved file stage karta hai
git merge --continue # Resolved merge complete karta hai
git merge --abort # In-progress merge cancel karne ki koshish karta hai
git mergetool # Configured merge tool se conflicts resolve karta hai

09. REBASE AND CHERRY-PICK

git rebase --continue # Conflict resolve aur stage karne ke baad rebase continue karta hai
git rebase --skip # Current replayed commit skip karta hai; changes omit ho sakte hain
git rebase --abort # Rebase cancel karke original branch state restore karta hai
git rebase --autostash main # Temporary stash ke saath rebase karta hai; apply conflict possible hai
git cherry-pick COMMIT_ID # Selected commit ka change current branch par apply karta hai
git cherry-pick -x COMMIT_ID # Original commit ID message mein record karta hai
git cherry-pick --no-commit COMMIT_ID # Change stage karta hai bina commit banaye
git cherry-pick --continue # Conflict resolution ke baad cherry-pick continue karta hai
git cherry-pick --abort # In-progress cherry-pick cancel karta hai

11. RESET, RESTORE, REVERT AND RECOVERY
git restore file.txt # WARNING: Unstaged tracked changes discard karta hai
git restore --source=HEAD~1 -- file.txt # Purane commit ka file content working tree mein laata hai
git restore --staged --worktree --source=HEAD file.txt # WARNING: Staged aur unstaged file edits discard karta hai
git reset --soft HEAD~1 # Branch tip peeche karta hai, index aur working tree preserve karta hai
git reset --mixed HEAD~1 # Branch tip aur index reset karta hai, working files preserve karta hai
git reset --hard HEAD~1 # WARNING: Branch, index aur tracked working changes reset karta hai
git revert COMMIT_ID # Existing commit ko undo karne wala naya commit banata hai
git revert --no-commit COMMIT_ID # Reverse change apply karta hai bina commit banaye
git revert -m 1 MERGE_COMMIT_ID # Merge commit ko first-parent baseline se revert karta hai
git revert --abort # In-progress revert cancel karta hai
git reflog # Local HEAD movements aur recovery candidates dikhata hai
git reflog show main # Main branch ke local ref movements dikhata hai
git branch rescue COMMIT_ID # Purane commit ko rescue branch se reachable banata hai
git clean -nd # Untracked files/directories removal ka preview dikhata hai
git clean -fd # WARNING: Untracked files aur directories delete karta hai
git clean -ndx # Ignored files samet deletion preview dikhata hai
git clean -fdx # WARNING: Ignored aur untracked data permanently delete kar sakta hai
git fsck --full # Repository object integrity check karta hai
git fsck --lost-found # Dangling objects recovery ke liye lost-found mein likhta hai
